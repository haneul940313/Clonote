const CONFIG = { notionVersion: '2022-06-28', timeZone: 'Asia/Seoul', pageSize: 100, props: { token: 'NOTION_TOKEN', permitDatabaseId: 'PERMIT_DATABASE_ID', protectionDatabaseId: 'PROTECTION_DATABASE_ID', defaultMonth: 'DEFAULT_MONTH', }, permitFields: { date: ['전표일자'], manager: ['담당자'], customer: ['거래처'], address: ['현장주소'], item: ['거래품목'], constructionCost: ['시공비'], totalAmount: ['총합계', '합계', '시공비'], paymentStatus: ['결제'], }, protectionFields: { date: ['전표일자'], manager: ['담당자'], customer: ['거래처'], address: ['현장주소'], item: ['거래품목'], constructionCost: ['시공비'], totalAmount: ['총합계', '합계', '시공비'], paymentStatus: ['결제'], }, }; function doGet(e) { try { const params = (e && e.parameter) || {}; const action = String(params.action || 'health').toLowerCase(); if (action === 'health') { return json_({ ok: true, message: 'Apps Script is running', actions: ['health', 'sites'], }); } if (action === 'sites') { return json_(loadSites_(params)); } throw new Error('알 수 없는 action입니다: ' + action); } catch (err) { return json_({ ok: false, error: String(err && err.message ? err.message : err), stack: String(err && err.stack ? err.stack : ''), }); } } function doPost(e) { return doGet(e); } function loadSites_(params) { const month = params.month || prop_('defaultMonth') || ''; const permitPages = queryAllDatabase_( prop_('permitDatabaseId'), buildMonthAndUnpaidFilter_(month, '전표일자', '결제') ); const protectionPages = queryAllDatabase_( prop_('protectionDatabaseId'), buildMonthAndUnpaidFilter_(month, '전표일자', '결제') ); const permitRows = permitPages.map(function(page) { return mapBillingRow_(page, CONFIG.permitFields, 'permit'); }); const protectionRows = protectionPages.map(function(page) { return mapBillingRow_(page, CONFIG.protectionFields, 'protection'); }); const rows = permitRows.concat(protectionRows) .filter(function(row) { return row.customer && row.content && row.totalAmount > 0 && row.paymentStatus.indexOf('미결제') > -1; }) .sort(function(a, b) { return String(a.slipDate || '').localeCompare(String(b.slipDate || '')) || String(a.customer || '').localeCompare(String(b.customer || ''), 'ko') || String(a.content || '').localeCompare(String(b.content || ''), 'ko'); }); return { ok: true, month: month || null, count: rows.length, sites: rows, }; } function buildMonthAndUnpaidFilter_(month, dateProperty, paymentProperty) { const filters = [ { property: paymentProperty, multi_select: { contains: '미결제', }, }, ]; if (month) { filters.push( { property: dateProperty, date: { on_or_after: month + '-01', }, }, { property: dateProperty, date: { before: nextMonth_(month) + '-01', }, } ); } return { and: filters }; } function mapBillingRow_(page, fields, source) { const props = page.properties || {}; const customer = readTextAny_(props, fields.customer); const manager = readTextAny_(props, fields.manager); const address = readTextAny_(props, fields.address); const itemText = readTextAny_(props, fields.item); const items = splitItems_(itemText); const paymentStatus = readMultiSelectText_(props, fields.paymentStatus); const constructionCost = readFirstPositiveNumber_(props, fields.constructionCost); const totalAmount = readFirstPositiveNumber_(props, fields.totalAmount); return { source: source, id: page.id, url: page.url || '', customer: customer, title: customer, content: customer, name: customer, slipDate: readDate_(props, fields.date), address: address, manager: manager, contacts: customer ? [ { id: 'customer:' + customer, name: customer, type: '거래처', phone: '', }, ] : [], contactIds: customer ? ['customer:' + customer] : [], hasContact: !!customer, items: items, itemText: itemText, constructionCost: constructionCost, totalAmount: totalAmount, paymentStatus: paymentStatus, paid: paymentStatus.indexOf('미결제') === -1, }; } function queryAllDatabase_(databaseId, filter) { if (!databaseId) throw new Error('데이터베이스 ID가 비어 있습니다.'); const results = []; let cursor = null; do { const payload = { page_size: CONFIG.pageSize, }; if (cursor) payload.start_cursor = cursor; if (filter) payload.filter = filter; const res = notionFetch_('/databases/' + normalizeId_(databaseId) + '/query', 'post', payload); (res.results || []).forEach(function(page) { results.push(page); }); cursor = res.has_more ? res.next_cursor : null; } while (cursor); return results; } function notionFetch_(path, method, payload) { const token = prop_('token'); if (!token) throw new Error('스크립트 속성 NOTION_TOKEN이 비어 있습니다.'); const options = { method: method || 'get', muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + token, 'Notion-Version': CONFIG.notionVersion, 'Content-Type': 'application/json', }, }; if (payload) { options.payload = JSON.stringify(payload); } const response = UrlFetchApp.fetch('https://api.notion.com/v1' + path, options); const status = response.getResponseCode(); const text = response.getContentText(); const data = text ? JSON.parse(text) : {}; if (status < 200 || status >= 300) { throw new Error('Notion API error ' + status + ': ' + (data.message || text)); } return data; } function prop_(key) { const propName = CONFIG.props[key] || key; return PropertiesService.getScriptProperties().getProperty(propName); } function normalizeId_(id) { return String(id || '') .replace(/^collection:\/\//, '') .replace(/-/g, '') .trim(); } function nextMonth_(yyyyMm) { const parts = String(yyyyMm).split('-'); const year = Number(parts[0]); const month = Number(parts[1]); const date = new Date(year, month, 1); return Utilities.formatDate(date, CONFIG.timeZone, 'yyyy-MM'); } function getProp_(props, names) { for (let i = 0; i < names.length; i++) { if (props[names[i]]) return props[names[i]]; } return null; } function readTextAny_(props, names) { const prop = getProp_(props, names); if (!prop) return ''; if (prop.title) { return prop.title.map(function(t) { return t.plain_text || ''; }).join('').trim(); } if (prop.rich_text) { return prop.rich_text.map(function(t) { return t.plain_text || ''; }).join('').trim(); } if (prop.select && prop.select.name) { return prop.select.name.trim(); } if (prop.multi_select) { return prop.multi_select.map(function(item) { return item.name; }).join(', ').trim(); } if (prop.formula) { if (typeof prop.formula.string === 'string') return prop.formula.string.trim(); if (typeof prop.formula.number === 'number') return String(prop.formula.number); } if (prop.phone_number) return prop.phone_number.trim(); if (prop.email) return prop.email.trim(); if (prop.url) return prop.url.trim(); return ''; } function readMultiSelectText_(props, names) { const prop = getProp_(props, names); if (!prop) return ''; if (prop.multi_select) { return prop.multi_select.map(function(item) { return item.name; }).join(', '); } if (prop.select && prop.select.name) { return prop.select.name; } if (prop.status && prop.status.name) { return prop.status.name; } return readTextAny_(props, names); } function readFirstPositiveNumber_(props, names) { for (let i = 0; i < names.length; i++) { const value = readNumber_(props, [names[i]]); if (value > 0) return value; } return 0; } function readNumber_(props, names) { const prop = getProp_(props, names); if (!prop) return 0; if (typeof prop.number === 'number') return prop.number; if (prop.formula) { if (typeof prop.formula.number === 'number') return prop.formula.number; if (typeof prop.formula.string === 'string') return toNumber_(prop.formula.string); } return toNumber_(readTextAny_(props, names)); } function readDate_(props, names) { const prop = getProp_(props, names); if (!prop || !prop.date || !prop.date.start) return ''; return prop.date.start; } function splitItems_(text) { return String(text || '') .split(',') .map(function(item) { return item.trim(); }) .filter(Boolean); } function toNumber_(value) { if (typeof value === 'number') return value; if (value === null || value === undefined || value === '') return 0; const cleaned = String(value).replace(/[^\d.-]/g, ''); const number = Number(cleaned); return isNaN(number) ? 0 : number; } function json_(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
